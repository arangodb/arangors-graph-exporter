64pub(crate) async fn get_all_shard_data(
65    db_config: &DatabaseConfiguration,
66    load_config: &DataLoadConfiguration,
67    shard_map: &ShardMap,
68    result_channels: Vec<tokio::sync::mpsc::Sender<Bytes>>,
69    deployment_type: &DeploymentType,
70    projections: Option<HashMap<String, Vec<String>>>,
71) -> Result<(), GraphLoaderError> {
72    let use_tls = db_config.endpoints[0].starts_with("https://");
73    let client_config = ClientConfig::builder()
74        .n_retries(5)
75        .use_tls(use_tls)
76        .tls_cert_opt(db_config.tls_cert.clone())
77        .build();
78    let client = build_client(&client_config)?;
79
80    // Start a single dump context on all involved dbservers, we can do
81    // this sequentially, since it is not performance critical, we can
82    // also use the same HTTP client and the same first endpoint:
83    let mut dbservers: Vec<DBServerInfo> = vec![];
84    let mut error_happened = false;
85    let mut error: String = "".into();
86    for (server, shard_list) in shard_map.iter() {
87        let url = if deployment_type == DeploymentType::Cluster {
88            make_url(db_config, &format!("/_api/dump/start?dbserver={}", server))
89        } else {
90            make_url(db_config, "/_api/dump/start")
91        };
92
93        let body_v: Vec<u8> = if projections.is_some() {
94            let body = DumpStartBodyWithProjections {
95                batch_size: load_config.batch_size,
96                prefetch_count: load_config.prefetch_count,
97                parallelism: load_config.parallelism,
98                shards: shard_list.clone(), // in case of a single server, this is a collection and not a shard
99                projections: projections.clone(),
100            };
101            serde_json::to_vec::<DumpStartBodyWithProjections>(&body)
102                .expect("could not serialize DumpStartBody")
103        } else {
104            let body = DumpStartBody {
105                batch_size: load_config.batch_size,
106                prefetch_count: load_config.prefetch_count,
107                parallelism: load_config.parallelism,
108                shards: shard_list.clone(), // in case of a single server, this is a collection and not a shard
109            };
110            serde_json::to_vec::<DumpStartBody>(&body).expect("could not serialize DumpStartBody")
111        };
112
113        let resp = handle_auth(client.post(url), db_config)
114            .body(body_v)
115            .send()
116            .await;
117        let r = handle_arangodb_response(resp, |c| {
118            c == StatusCode::NO_CONTENT || c == StatusCode::OK || c == StatusCode::CREATED
119        })
120        .await;
121        if let Err(rr) = r {
122            error = rr;
123            error_happened = true;
124            break;
125        }
126        let r = r.unwrap();
127        let headers = r.headers();
128        if let Some(id) = headers.get("X-Arango-Dump-Id")
129            && let Ok(id) = id.to_str()
130        {
131            dbservers.push(DBServerInfo {
132                dbserver: server.clone(),
133                dump_id: id.to_owned(),
134            });
135        }
136        debug!("Started dbserver {}", server);
137    }
138
139    let client_clone_for_cleanup = client.clone();
140    let cleanup = |dbservers: Vec<DBServerInfo>| async move {
141        debug!("Doing cleanup...");
142        for dbserver in dbservers.iter() {
143            let url = if deployment_type == DeploymentType::Cluster {
144                make_url(
145                    db_config,
146                    &format!(
147                        "/_api/dump/{}?dbserver={}",
148                        dbserver.dump_id, dbserver.dbserver
149                    ),
150                )
151            } else {
152                make_url(db_config, &format!("/_api/dump/{}", dbserver.dump_id))
153            };
154            let resp = handle_auth(client_clone_for_cleanup.delete(url), db_config)
155                .send()
156                .await;
157            let r =
158                handle_arangodb_response(resp, |c| c == StatusCode::OK || c == StatusCode::CREATED)
159                    .await;
160            if let Err(rr) = r {
161                eprintln!(
162                    "An error in cancelling a dump context occurred, dbserver: {}, error: {}",
163                    dbserver.dbserver, rr
164                );
165                // Otherwise ignore the error, this is just a cleanup!
166            }
167        }
168    };
169
170    if error_happened {
171        // We need to cancel all dump contexts which we did get successfully:
172        cleanup(dbservers).await;
173        return Err(errors::GraphLoaderError::Other(error));
174    }
175
176    // We want to start the same number of tasks for each dbserver, each of
177    // them will send next requests until no more data arrives
178
179    #[derive(Debug)]
180    struct TaskInfo {
181        dbserver: DBServerInfo,
182        current_batch_id: u64,
183        last_batch_id: Option<u64>,
184        id: u64,
185    }
186
187    if dbservers.is_empty() {
188        // This actually happened writing integration tests, we cannot divide by zero
189        error!("No dbserver found. List is empty.");
190        return Err(GraphLoaderError::NoDatabaseServers);
191    }
192
193    let par_per_dbserver = (load_config.parallelism as usize).div_ceil(dbservers.len());
194    let mut task_set = JoinSet::new();
195    let mut endpoints_round_robin: usize = 0;
196    let mut consumers_round_robin: usize = 0;
197    for i in 0..par_per_dbserver {
198        for dbserver in &dbservers {
199            let mut task_info = TaskInfo {
200                dbserver: dbserver.clone(),
201                current_batch_id: i as u64,
202                last_batch_id: None,
203                id: i as u64,
204            };
205            //let client_clone = client.clone(); // the clones will share
206            //                                   // the connection pool
207            let use_tls = db_config.endpoints[0].starts_with("https://");
208            let client_config = ClientConfig::builder()
209                .n_retries(5)
210                .use_tls(use_tls)
211                .tls_cert_opt(db_config.tls_cert.clone())
212                .build();
213            let client_clone = build_client(&client_config)?;
214            let endpoint_clone = db_config.endpoints[endpoints_round_robin].clone();
215            endpoints_round_robin += 1;
216            if endpoints_round_robin >= db_config.endpoints.len() {
217                endpoints_round_robin = 0;
218            }
219            let database_clone = db_config.database.clone();
220            let result_channel_clone = result_channels[consumers_round_robin].clone();
221            consumers_round_robin += 1;
222            if consumers_round_robin >= result_channels.len() {
223                consumers_round_robin = 0;
224            }
225            let db_config_clone = db_config.clone();
226            task_set.spawn(async move {
227                loop {
228                    let mut url = format!(
229                        "{}/_db/{}/_api/dump/next/{}?dbserver={}&batchId={}",
230                        endpoint_clone,
231                        database_clone,
232                        task_info.dbserver.dump_id,
233                        task_info.dbserver.dbserver,
234                        task_info.current_batch_id
235                    );
236                    if let Some(last) = task_info.last_batch_id {
237                        url.push_str(&format!("&lastBatch={}", last));
238                    }
239                    let start = SystemTime::now();
240                    debug!(
241                        "{:?} Sending post request... {} {} {}",
242                        start,
243                        task_info.id,
244                        task_info.dbserver.dbserver,
245                        task_info.current_batch_id
246                    );
247                    let resp = handle_auth(client_clone.post(url), &db_config_clone)
248                        .send()
249                        .await;
250                    let resp = handle_arangodb_response(resp, |c| {
251                        c == StatusCode::OK || c == StatusCode::NO_CONTENT
252                    })
253                    .await?;
254                    let end = SystemTime::now();
255                    let dur = end.duration_since(start).unwrap();
256                    if resp.status() == StatusCode::NO_CONTENT {
257                        // Done, cleanup will be done later
258                        debug!(
259                            "{:?} Received final post response... {} {} {} {:?}",
260                            end,
261                            task_info.id,
262                            task_info.dbserver.dbserver,
263                            task_info.current_batch_id,
264                            dur
265                        );
266                        return Ok::<(), String>(());
267                    }
268                    // Now the result was OK and the body is JSONL
269                    task_info.last_batch_id = Some(task_info.current_batch_id);
270                    task_info.current_batch_id += par_per_dbserver as u64;
271                    let body = resp
272                        .bytes()
273                        .await
274                        .map_err(|e| format!("Error in body: {:?}", e))?;
275                    match result_channel_clone.send(body).await {
276                        Ok(_) => {
277                            // Successfully sent to the channel
278                            debug!("Successfully sent message to channel.");
279                        }
280                        Err(e) => {
281                            panic!("Failed to send to channel: {:?}", e);
282                        }
283                    }
284                }
285            });
286        }
287    }
288    while let Some(res) = task_set.join_next().await {
289        match res {
290            Ok(Ok(_x)) => {
291                debug!("Got OK result!");
292            }
293            Ok(Err(err_msg)) => {
294                // This handles errors that occur within the task itself
295                panic!("Task failed with error: {:?}", err_msg);
296            }
297            Err(join_err) => {
298                // This handles errors in the task joining process (e.g., task panicked)
299                panic!("Task join error occurred: {:?}", join_err)
300            }
301        }
302    }
303    cleanup(dbservers).await;
304    debug!("Done cleanup and channel is closed!");
305    Ok(())
306    // We drop the result_channel when we leave the function.
307}

309pub(crate) fn compute_faked_shard_map(coll_list: &[String]) -> ShardMap {
310    // not really faked, but to be able to implement this quickly, we need to
311    // simply build a map exposing the collection names instead of shard names.
312    let mut result: ShardMap = HashMap::new();
313    for c in coll_list.iter() {
314        result.insert(c.clone().to_string(), vec![c.clone()]);
315    }
316    result
317}

319pub(crate) fn compute_shard_map(
320    sd: &ShardDistribution,
321    coll_list: &[String],
322) -> Result<ShardMap, String> {
323    let mut result: ShardMap = HashMap::new();
324    for c in coll_list.iter() {
325        // Handle the case of a smart edge collection. If c is
326        // one, then we also find a collection called `_to_`+c.
327        // In this case, we must not get those shards, because their
328        // data is already contained in `_from_`+c, just sharded
329        // differently.
330        let mut ignore: HashSet<String> = HashSet::new();
331        let smart_name = "_to_".to_owned() + c;
332        match sd.results.get(&smart_name) {
333            None => (),
334            Some(coll_dist) => {
335                // Keys of coll_dist are the shards, value has leader:
336                for shard in coll_dist.plan.keys() {
337                    ignore.insert(shard.clone());
338                }
339            }
340        }
341        match sd.results.get(c) {
342            None => {
343                return Err(format!("collection {} not found in shard distribution", c));
344            }
345            Some(coll_dist) => {
346                // Keys of coll_dist are the shards, value has leader:
347                for (shard, location) in &(coll_dist.plan) {
348                    if !ignore.contains(shard) {
349                        let leader = &(location.leader);
350                        match result.get_mut(leader) {
351                            None => {
352                                result.insert(leader.clone(), vec![shard.clone()]);
353                            }
354                            Some(list) => {
355                                list.push(shard.clone());
356                            }
357                        }
358                    }
359                }
360            }
361        }
362    }
363    Ok(result)
